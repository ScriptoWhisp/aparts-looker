    ### Inline `<script>` block (before external script tags)

    Define the shared state object and helpers:

    ```
    window.state = {
      properties: [],
      pending: [],
      rejected: [],
      checklists: {},
      priceHistory: {},
      compareSelected: new Set(),
      isochroneGeoJson: null,
      districtsGeoJson: null,
      districtsData: []
    };
    window.mapFilter = "all";

    window.escapeHtml = function(s) {
      return String(s == null ? "" : s).replace(/[&<>"']/g, function(c) {
        return {"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[c];
      });
    };
    window.fmtEur = function(n) {
      n = Number(n) || 0;
      return n.toLocaleString("et-EE", {maximumFractionDigits: 0}) + " €";
    };
    window.scoreColor = function(score) {
      if (score >= 75) return "#10b981";
      if (score >= 50) return "#f59e0b";
      return "#ef4444";
    };
    window.commuteTier = function(mins) {
      if (mins == null) return null;
      if (mins < 15) return "green";
      if (mins < 25) return "amber";
      return "red";
    };
    ```
